## 5. 数据模型

### 5.1 数据库表

```sql
-- 客户表
CREATE TABLE customers (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    keywords TEXT,
    git_repo TEXT,
    contact TEXT,
    created_at DATETIME,
    updated_at DATETIME
);

-- 项目表
CREATE TABLE projects (
    id INTEGER PRIMARY KEY,
    customer_id INTEGER,
    name TEXT NOT NULL,
    keywords TEXT,
    status TEXT,
    progress INTEGER,
    git_repo TEXT,
    oc_collab_enabled BOOLEAN,
    created_at DATETIME,
    updated_at DATETIME,
    FOREIGN KEY (customer_id) REFERENCES customers(id)
);

-- 处理日志表
CREATE TABLE processing_logs (
    id INTEGER PRIMARY KEY,
    input_type TEXT,
    content TEXT,
    customer_id INTEGER,
    project_id INTEGER,
    result TEXT,
    created_at DATETIME,
    FOREIGN KEY (customer_id) REFERENCES customers(id),
    FOREIGN KEY (project_id) REFERENCES projects(id)
);
```
